'use client'

import { useState, useEffect, useMemo, useRef, useCallback } from 'react'
import Link from 'next/link'
import { getAllCharacters } from '@/data/characters'
import { recordJjkdleSession } from '@/lib/progress'
import { CheckCircle, Clock, Home, Flame, Calendar } from 'lucide-react'
import { displayPowerLevel } from '@/types/character'
import type { JJKCharacter } from '@/types/character'
import {
  seededRandom, todayStr, getTimeUntilReset,
  getDayResult, saveDayResult, isDayCompleted,
  getDailyStreakData, updateDailyStreak,
  buildDayOptions,
  type DailyResult, type DailyStreakData, type DayOption,
} from '@/lib/daily-challenge'

/* â”€â”€ types â”€â”€ */
interface GuessEntry { character: JJKCharacter; result: AttributeResult }
interface AttributeResult { species: boolean; gender: boolean; grade: 'match' | 'partial' | 'none'; affiliation: boolean; clan: boolean; arc: boolean; powerDir: 'up' | 'down' | 'equal' | 'unknown' }

const GAME_ID = 'jjkdle'

/* Grade hierarchy for partial matching */
const GRADE_ORDER = ['special grade','special grade 1','grade 1','semi-grade 1','grade 2','semi-grade 2','grade 3','grade 4','ungraded']

/* â”€â”€ Daily target for any date â”€â”€ */
function getTargetForDate(all: JJKCharacter[], date: string): JJKCharacter {
  const seed = date.split('-').reduce((acc, part) => acc + parseInt(part), 0) * 9973
  return all[Math.floor(seededRandom(seed) * all.length)]
}

/* â”€â”€ page â”€â”€ */
export default function JJKdlePage() {
  const all = useMemo(() => getAllCharacters(), [])

  const [activeDate, setActiveDate] = useState(() => todayStr())
  const [target, setTarget] = useState<JJKCharacter | null>(null)
  const [input, setInput] = useState('')
  const [filtered, setFiltered] = useState<JJKCharacter[]>([])
  const [guesses, setGuesses] = useState<GuessEntry[]>([])
  const [gameState, setGameState] = useState<'playing' | 'won'>('playing')
  const [showHelp, setShowHelp] = useState(false)
  const [revealAffiliation, setRevealAffiliation] = useState(false)
  const [revealPhoto, setRevealPhoto] = useState(false)
  const [revealInitial, setRevealInitial] = useState(false)
  const recordedRef = useRef(false)

  const [dailyStreak, setDailyStreak] = useState<DailyStreakData | null>(null)
  const [countdown, setCountdown] = useState('')
  const [dayOptions, setDayOptions] = useState<DayOption[]>([])
  const [showDayPicker, setShowDayPicker] = useState(false)

  useEffect(() => { const u = () => setCountdown(getTimeUntilReset()); u(); const i = setInterval(u, 1000); return () => clearInterval(i) }, [])

    const gG = (guess.grade || '').toLowerCase(); const tG = (t.grade || '').toLowerCase()
    if (gG === tG) gradeStatus = 'match'
    else { const gI = GRADE_ORDER.indexOf(gG); const tI = GRADE_ORDER.indexOf(tG); if (gI !== -1 && tI !== -1 && Math.abs(gI - tI) === 1) gradeStatus = 'partial' }

    return { species: (guess.species || '') === (t.species || ''), gender: (guess.gender || '') === (t.gender || ''), grade: gradeStatus, affiliation: (guess.affiliation || '') === (t.affiliation || ''), clan: (guess.clan || '') === (t.clan || ''), arc: (guess.arc || '') === (t.arc || ''), powerDir }
  }

    if (!target || gameState !== 'playing' || guessedIds.has(c.id) || dayCompleted) return

    const newGuesses = [{ character: c, result: res }, ...guesses]

    if (c.id === target.id) {

    }

  const switchDay = (date: string) => { setShowDayPicker(false); initGame(date) }

      <span className="text-[11px] font-semibold text-center leading-tight line-clamp-2">{value || 'â€”'}</span>

  }

    if (dir === 'up') { color = 'bg-green-600 text-white'; icon = 'â¬†ï¸' } else if (dir === 'down') { color = 'bg-red-600 text-white'; icon = 'â¬‡ï¸' } else if (dir === 'equal') { color = 'bg-green-700 text-white'; icon = 'âœ…' }

      <div className="min-h-screen text-purple-100 flex flex-col relative">
        <div aria-hidden className="pointer-events-none fixed inset-0 -z-10 bg-gradient-to-b from-black/65 via-black/55 to-black/70" />

        {/* Header */}

                <Link href="/" className="flex items-center gap-1 text-purple-300/70 hover:text-purple-100 transition-colors"><Home className="w-4 h-4" /><span className="hidden sm:inline">Home</span></Link>

                  <span className="text-[10px] px-2 py-0.5 bg-fuchsia-500/20 border border-fuchsia-500/50 text-fuchsia-400 rounded-full font-bold ml-2">DAILY</span>

            </div>

          {/* Completed Banner */}
          {dayCompleted && dayResult && target && (
            <div className="mb-8 max-w-md mx-auto">
              <div className="relative rounded-2xl p-6 text-center overflow-hidden" style={{ background: 'linear-gradient(180deg, rgba(12, 5, 32, 0.95) 0%, rgba(20, 10, 50, 0.98) 100%)', border: '2px solid rgba(168, 85, 247, 0.5)' }}>
                <div className="absolute top-2 left-2 w-6 h-6 border-l-2 border-t-2 border-purple-400/60" />
                <div className="absolute top-2 right-2 w-6 h-6 border-r-2 border-t-2 border-purple-400/60" />

'opacity-40 grayscale pointer-events-none' : 'cursor-pointer hover:brightness-110 hover:scale-[1.02]'}`}><span className="text-4xl sm:text-5xl">ðŸ«</span></div>

          )}

          {/* Win Card */}
          {gameState === 'won' && target && !dayCompleted && (
            <div className="max-w-sm mx-auto mb-6 relative rounded-2xl p-6 text-center overflow-hidden border-2 border-green-500/50" style={{ background: 'linear-gradient(145deg, rgba(22, 101, 52, 0.3) 0%, rgba(12, 5, 32, 0.95) 100%)', animation: 'fadeIn 0.4s ease-out' }}>
              <div className="absolute top-2 left-2 w-4 h-4 border-l-2 border-t-2 border-green-400/60" /><div className="absolute top-2 right-2 w-4 h-4 border-r-2 border-t-2 border-green-400/60" /><div className="absolute bottom-2 left-2 w-4 h-4 border-l-2 border-b-2 border-green-400/60" /><div className="absolute bottom-2 right-2 w-4 h-4 border-r-2 border-b-2 border-green-400/60" />
              <CheckCircle className="w-10 h-10 mx-auto mb-1 text-green-400" />
              <div className="text-2xl sm:text-3xl font-bold text-green-400 mb-2">Correct!</div>
              <div className="flex items-center justify-center gap-3 mb-3"><img src={target.image_url} alt={target.name} className="w-14 h-14 sm:w-16 sm:h-16 object-cover object-top rounded-lg bg-purple-950/50 p-0.5 border border-purple-600/50" /><div className="text-lg sm:text-xl font-bold text-white">{target.name}</div></div>
              <div className="text-sm text-purple-300">Guessed in {guesses.length} attempt{guesses.length !== 1 ? 's' : ''}</div>
              {dailyStreak && dailyStreak.currentStreak > 0 && activeDate === todayStr() && <div className="mt-3 flex items-center justify-center gap-2 text-amber-400"><Flame className="w-5 h-5" /><span className="font-bold">{dailyStreak.currentStreak} day streak</span>{dailyStreak.currentStreak === dailyStreak.bestStreak && dailyStreak.currentStreak > 1 && <span className="text-xs bg-amber-400/20 px-2 py-0.5 rounded-full">Best!</span>}</div>}
              {activeDate === todayStr() && <div className="mt-2 flex items-center justify-center gap-2 text-purple-400 text-sm"><Clock className="w-4 h-4" /><span>Next daily in {countdown}</span></div>}
            </div>
          )}
        </div>
      </div>
      <style jsx>{`
        @keyframes attrIn { from { opacity:0; transform:translateX(-10px); } to { opacity:1; transform:translateX(0); } }
        .attr-reveal { opacity:0; animation: attrIn .35s ease-out forwards; }
        @keyframes fadeIn { from { opacity: 0; transform: scale(0.95); } to { opacity: 1; transform: scale(1); } }
      `}</style>
    </>
  )
}
